def physical_cleanup_fixture(case, record):
    directory = tempfile.TemporaryDirectory(prefix='optional-current-owner-')
    case.addCleanup(directory.cleanup)
    root = Path(directory.name)
    paths = {cleanup.MIGRATION, *cleanup.CHECKER_PATHS, *cleanup.SOURCE_PATHS,
             *record['inventory'], str(sources.PINS), str(guards.PINS)}
    for group in cleanup.BOUND_GROUPS:
        paths.update(record[group])
    for paths_in_group in sources.GROUP_PATHS.values():
        paths.update(paths_in_group)
    for name in paths:
        target = root/name
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(ROOT/name, target)
    return root
