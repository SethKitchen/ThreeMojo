    def test_rounded_consumers_keep_variant_resolver_witness_and_census_live(self):
        import rounded_line_selector_contracts as selector
        import rounded_line_singleton_contracts as singleton
        path = self.root/CENTER_LANE
        original = path.read_bytes()
        current = original.decode()
        mutations = (
            (CENTER_LANE, current.replace('from std.utils import Variant',
                                          'from std.utils import Variant as Hidden', 1).encode()),
            (CENTER_LANE, current.replace('from std.utils import Variant',
                                          'from unreviewed import Variant', 1).encode()),
            (CENTER_WITNESS, (self.root/CENTER_WITNESS).read_bytes() + b'\n# Changed witness.\n'),
            ('std/utils.mojo', b'struct Variant: pass\n'),
            ('fresh_namespace/center_consumer.mojo',
             b'from extensions.carla.lane_refinement import _try_lane_center_bound as hidden\n'),
        )
        for contract in (selector, singleton):
            contract.verify(self.root)
        for name, changed in mutations:
            target = self.root/name
            previous = target.read_bytes() if target.exists() else None
            self.assertNotEqual(changed, previous)
            created_directories = []
            parent = target.parent
            while not parent.exists():
                created_directories.append(parent)
                parent = parent.parent
            target.parent.mkdir(parents=True, exist_ok=True)
            with self.subTest(live_input=name, mutation=changed[:80]):
                try:
                    target.write_bytes(changed)
                    for contract in (selector, singleton):
                        with self.subTest(contract=contract.__name__):
                            with self.assertRaises(ValueError):
                                contract.verify_premises(self.root)
                            with self.assertRaises(ValueError):
                                contract.verify(self.root)
                finally:
                    if previous is None:
                        target.unlink()
                    else:
                        target.write_bytes(previous)
                    for directory in created_directories:
                        directory.rmdir()
                for contract in (selector, singleton):
                    contract.verify(self.root)
        self.assertEqual(path.read_bytes(), original)
