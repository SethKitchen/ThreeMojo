struct Moving(Boolable, Movable):
    var value: Bool

    def __init__(out self, value: Bool):
        self.value = value

    def __init__(out self, *, deinit move: Self):
        self.value = move.value
        print("MOVE")

    def __bool__(self) -> Bool:
        print("CONVERT")
        return self.value


def main():
    if Moving(False) or Moving(True):
        print("T")
    else:
        print("F")
