struct CopyOnly(ImplicitlyCopyable):
    var index: Int

    def __init__(out self, index: Int):
        self.index = index
        print("CREATE", index)

    def __init__(out self, *, copy: Self):
        self.index = copy.index
        print("COPY", self.index)


def optional(index: Int, full: Bool) -> Optional[CopyOnly]:
    print("MAKE", index, full)
    if full:
        return CopyOnly(index)
    return None


def retained(first: Bool, second: Bool):
    print("RETAINED", first, second)
    var left = optional(0, first)
    var right = optional(1, second)
    if left and right:
        print("BODY T")
    else:
        print("BODY F")
    print("RETAINED AFTER", Bool(left), Bool(right))


def temporary(first: Bool, second: Bool):
    print("TEMPORARY", first, second)
    if optional(0, first) and optional(1, second):
        print("BODY T")
    else:
        print("BODY F")


def main():
    for first in [False, True]:
        for second in [False, True]:
            retained(first, second)
            temporary(first, second)
