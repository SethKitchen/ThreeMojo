struct Logged(ImplicitlyCopyable):
    var name: Int

    def __init__(out self, name: Int):
        self.name = name

    def __init__(out self, *, copy: Self):
        self.name = copy.name
        print("COPY")


def main():
    var empty = Optional[Logged](None)
    var full = Optional[Logged](Logged(1))
    if empty or full:
        print("T")
    else:
        print("F")
    print(full.value().name)
