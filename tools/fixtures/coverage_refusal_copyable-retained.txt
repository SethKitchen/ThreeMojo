struct Copied(ImplicitlyCopyable, Boolable):
    var value: Bool

    def __init__(out self, value: Bool):
        self.value = value

    def __init__(out self, *, copy: Self):
        self.value = copy.value
        print("COPY")

    def __bool__(self) -> Bool:
        print("CONVERT")
        return self.value


def main():
    var copied = Copied(True)
    var other = Copied(False)
    if copied or other:
        print("T")
    else:
        print("F")
