struct State(Movable):
    var log: String
    var mask: Int
    var fail: Int
    var changing: Bool
    var counts: List[Int]

    def __init__(out self, mask: Int, fail: Int, changing: Bool):
        self.log = String("")
        self.mask = mask
        self.fail = fail
        self.changing = changing
        self.counts = List[Int](length=8, fill=0)

    def enter(mut self, index: Int) raises:
        self.log += "+" + String(index)
        if index == self.fail:
            raise Error("fixture exception")

    def convert(mut self, index: Int) -> Bool:
        self.log += String(index)
        var count = self.counts[index]
        self.counts[index] = count + 1
        var value = (self.mask & (1 << index)) != 0
        return value != (self.changing and count % 2 == 1)


comptime Cell = Pointer[State, MutUntrackedOrigin]


trait Made(Boolable, Movable, Deinitable):
    @staticmethod
    def make(index: Int, state: Cell) -> Self:
        ...


struct Probe(Made):
    var index: Int
    var state: Cell

    def __init__(out self, index: Int, state: Cell):
        self.index = index
        self.state = state

    @staticmethod
    def make(index: Int, state: Cell) -> Self:
        return Self(index, state)

    def __bool__(self) -> Bool:
        return self.state[].convert(self.index)

    def __deinit__(deinit self):
        self.state[].log += "~" + String(self.index)

struct Shared(ImplicitlyCopyable, Boolable):
    var index: Int
    var state: Cell

    def __init__(out self, index: Int, state: Cell):
        self.index = index
        self.state = state

    def __init__(out self, *, copy: Self):
        self.index = copy.index
        self.state = copy.state
        self.state[].log += "=" + String(self.index)

    def __bool__(self) -> Bool:
        return self.state[].convert(self.index)

    def __deinit__(deinit self):
        self.state[].log += "~" + String(self.index)


struct Moving(Made):
    var index: Int
    var state: Cell

    @staticmethod
    def make(index: Int, state: Cell) -> Self:
        return Self(index, state)

    def __init__(out self, index: Int, state: Cell):
        self.index = index
        self.state = state

    def __init__(out self, *, deinit move: Self):
        self.index = move.index
        self.state = move.state
        self.state[].log += ">" + String(self.index)

    def __bool__(self) -> Bool:
        return self.state[].convert(self.index)

    def __deinit__(deinit self):
        self.state[].log += "~" + String(self.index)


@fieldwise_init
struct Bits(ImplicitlyCopyable, Boolable):
    var index: Int
    var state: Cell

    def __bool__(self) -> Bool:
        return self.state[].convert(self.index)


def _cov_eval_truth(index: Int, state: Cell) raises -> Bits:

def _cov_eval_buffer(index: Int, state: Cell) raises -> Moving:

def _cov_eval_leaf(index: Int, state: Cell) raises -> Probe:

def _cov_eval_node(index: Int, state: Cell) raises -> Other:

def _cov_branch(index: Int, state: Cell) raises -> Bool:
    state[].enter(index)
    return (state[].mask & (1 << index)) != 0
