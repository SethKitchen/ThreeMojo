struct Left(Movable, Boolable):
    var value: Bool

    def __init__(out self, value: Bool):
        self.value = value
        print("MAKE L", value)

    def __bool__(self) -> Bool:
        print("BOOL L", self.value)
        return self.value

    def __merge_with__[other_type: AnyType](self) -> Bool:
        print("MERGE L", self.value)
        return not self.value


struct Right(Movable, Boolable):
    var value: Bool

    def __init__(out self, value: Bool):
        self.value = value
        print("MAKE R", value)

    def __bool__(self) -> Bool:
        print("BOOL R", self.value)
        return self.value

    def __merge_with__[other_type: AnyType](self) -> Bool:
        print("MERGE R", self.value)
        return not self.value


def main():
    for first in [False, True]:
        for second in [False, True]:
            print("CASE OR", first, second)
            if Left(first) or Right(second):
                print("BODY T")
            else:
                print("BODY F")
